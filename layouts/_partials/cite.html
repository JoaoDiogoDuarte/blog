{{- /* Turns citations like [HR04] or [GM84, KL20] into links to the matching reference. A reference is a list item that starts with its key, e.g. "- [HR04] C.-Y. Hsiao, ..."; it gets id="ref-HR04". The entry's own bracket is written as [ so the citation pass below skips it. */ -}} {{- $key := `[A-Z][A-Za-z]*\+?\d{2}` -}} {{- $c := replaceRE (printf `
  • (\s*

    )?\[(%s)\]` $key) `

  • $1[$2]` . -}} {{- /* Each pass links every other key of a multi-key citation, so repeat. */ -}} {{- range seq 4 -}} {{- $c = replaceRE (printf `(\[|, )(%s)(\]|,)` $key) `$1$2$3` $c -}} {{- end -}} {{- $c | safeHTML -}}
